(function(){
    const posts=document.querySelectorAll('.post_body');
    let html='';
    for(let i=0;i<posts.length;i++){
        html+=posts[i].outerHTML;
    }
    console.log(`Found ${posts.length} posts, sending...`);
    fetch('https://metabeys.meta-plutonian.workers.dev/submit',{method:'POST',body:html}).then(res=>res.text()).then(data=>console.log('Success:',data)).catch(err=>console.error('Failed:',err));
})();
